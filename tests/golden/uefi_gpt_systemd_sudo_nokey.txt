  test "$(findmnt -rn -o SOURCE -T /mnt/gentoo)" = /dev/sdc3
swapon --show=NAME --noheadings | awk -v expected=/dev/sdc2 '$1 == expected {found=1} END {exit found ? 0 : 1}'
  printf %s 'label: gpt
unit: sectors
/dev/sdc1 : size=512M, type=U
/dev/sdc2 : size=2048M, type=S
/dev/sdc3 : type=L
' | sfdisk --no-reread --force "$disk"
else
  wipefs -a -f "$disk"
  printf %s 'label: gpt
unit: sectors
/dev/sdc1 : size=512M, type=U
/dev/sdc2 : size=2048M, type=S
/dev/sdc3 : type=L
' | sfdisk --wipe always "$disk"
fi
partprobe "$disk" || true
udevadm settle || true
lsblk -o NAME,PATH,SIZE,TYPE,FSTYPE,MOUNTPOINTS "$disk"

set -eu

mkdir -p "$mount_root"

swap_device=/dev/sdc2

mkdir -p "$mount_root/boot"
if ! blkid -o value -s TYPE "$efi_device" 2>/dev/null | grep -Eq 'vfat|fat32'; then mkfs.vfat -F 32 "$efi_device"; fi
if ! findmnt -rn -T "$mount_root/boot" >/dev/null; then mount "$efi_device" "$mount_root/boot"; fi
findmnt -D "$mount_root"
