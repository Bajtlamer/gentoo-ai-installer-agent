umount -R /mnt/gentoo 2>/dev/null || true
for dev in $(lsblk -nr -o PATH "$disk"); do
  swapoff "$dev" 2>/dev/null || true
  umount "$dev" 2>/dev/null || true
done
if lsblk -nr -o MOUNTPOINTS "$disk" | grep -q '[^[:space:]]' || swapon --show=NAME --noheadings | awk -v disk="$disk" 'index($1, disk) == 1 {found=1} END {exit found ? 0 : 1}'; then
  printf '%s\n' "$disk is still mounted or used as swap and could not be released" >&2
  exit 1
fi
